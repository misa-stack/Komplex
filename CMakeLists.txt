cmake_minimum_required(VERSION 3.21)
project(Komplex VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Test)
add_library(fractal_core fractal.cpp renderer.cpp)
target_link_libraries(fractal_core PUBLIC Qt6::Widgets)
target_include_directories(fractal_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(fractal_ui viewer.cpp window.cpp lessons.cpp)
target_link_libraries(fractal_ui PUBLIC fractal_core)
add_executable(program main.cpp)
target_link_libraries(program PRIVATE fractal_ui)
foreach(target fractal_core fractal_ui program)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_executable(fractal_tests tests.cpp)
target_link_libraries(fractal_tests PRIVATE fractal_ui Qt6::Test)
add_test(NAME fractal_tests COMMAND fractal_tests)
set_tests_properties(fractal_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
